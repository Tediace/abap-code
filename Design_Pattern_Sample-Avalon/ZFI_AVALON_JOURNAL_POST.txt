CLASS lcl_posting_repo IMPLEMENTATION.


  METHOD zif_repository~create_by_ref.
    CONSTANTS : lc_doc_itm  TYPE tdobject VALUE 'DOC_ITEM',
                lc_txt_id   TYPE tdid VALUE '0001',
                lc_language TYPE spras VALUE 'E',
                lc_mwskz_a0 TYPE bseg-mwskz VALUE 'A0'.


    DATA li_return TYPE STANDARD TABLE OF bapiret2 WITH EMPTY KEY.
    DATA li_return_post TYPE STANDARD TABLE OF bapiret2 WITH EMPTY KEY.
    DATA li_tdline TYPE STANDARD TABLE OF tline WITH EMPTY KEY.
    DATA lwa_hdr_long_text  TYPE thead.
*    DATA lwa_log_h TYPE zfava_log_h.
    DATA lv_message TYPE bapi_msg.
    DATA lv_type TYPE bapi_mtype.
    DATA: lv_objtype TYPE awtyp,
          lv_objkey  TYPE awkey,
          lv_objsys  TYPE awsys.

    FIELD-SYMBOLS <source> TYPE ty_bapi_doc_post.
    ASSIGN im_data->* TO <source>.

    FIELD-SYMBOLS <result> TYPE zcl_ava_log_repository=>ty_source.

    "Check Document
    CALL FUNCTION 'BAPI_ACC_DOCUMENT_CHECK'
      EXPORTING
        documentheader    = <source>-document_header
      TABLES
        accountgl         = <source>-t_accountgl
        accountreceivable = <source>-t_receivable
        accountpayable    = <source>-t_payable
        accounttax        = <source>-t_tax
        currencyamount    = <source>-t_currency
        criteria          = <source>-t_criteria
        extension1        = <source>-t_extension
        extension2        = <source>-t_extension2
        return            = li_return.

    READ TABLE li_return TRANSPORTING NO FIELDS WITH KEY type = c_message_e.
    IF sy-subrc = 0.
      LOOP AT li_return INTO DATA(lwa_return) WHERE type = c_message_e.
        CONCATENATE lv_message lwa_return-message INTO lv_message SEPARATED BY space.
      ENDLOOP.
      lv_type = c_message_e.
    ELSE.
      lv_type = c_message_s.
    ENDIF.

    IF <source>-execute = abap_true.
      READ TABLE li_return TRANSPORTING NO FIELDS WITH KEY type = c_message_e.
      IF sy-subrc NE 0.
        CALL FUNCTION 'BAPI_ACC_DOCUMENT_POST'
          EXPORTING
            documentheader    = <source>-document_header
          IMPORTING
            obj_type          = lv_objtype
            obj_key           = lv_objkey
            obj_sys           = lv_objsys
          TABLES
            accountgl         = <source>-t_accountgl
            accountreceivable = <source>-t_receivable
            accountpayable    = <source>-t_payable
            accounttax        = <source>-t_tax
            currencyamount    = <source>-t_currency
            criteria          = <source>-t_criteria
            extension1        = <source>-t_extension
            extension2        = <source>-t_extension2
            return            = li_return_post.

        READ TABLE li_return_post TRANSPORTING NO FIELDS WITH KEY type = c_message_e.
        IF sy-subrc = 0.
          LOOP AT li_return_post INTO DATA(lwa_return_post) WHERE type = c_message_e.
            CONCATENATE lv_message lwa_return_post-message INTO lv_message SEPARATED BY space.
          ENDLOOP.
          lv_type = c_message_e.
        ELSE.
          CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
            EXPORTING
              wait = 'X'.

          lv_type = c_message_s.

          " Start of add by Triadi at 16.03.2021 GD1K915520
          " Set Long Text after document number is created
          LOOP AT <source>-t_data_detail INTO DATA(lwa_detail) WHERE long_text IS NOT INITIAL.

            READ TABLE li_return_post INTO lwa_return_post INDEX 1.
            IF sy-subrc = 0.
              lwa_hdr_long_text-tdname = <source>-document_header-comp_code && lwa_return_post-message_v2(10) &&
                                         lwa_return_post-message_v2+14(4) && lwa_detail-buzei.
            ENDIF.

            lwa_hdr_long_text-tdobject = lc_doc_itm.
            lwa_hdr_long_text-tdid = lc_txt_id.
            lwa_hdr_long_text-tdspras = lc_language.

            "Set Long Text
            li_tdline = VALUE #( BASE li_tdline ( tdline = lwa_detail-long_text ) ).


            CALL FUNCTION 'SAVE_TEXT'
              EXPORTING
                client          = sy-mandt
                header          = lwa_hdr_long_text
                savemode_direct = abap_true
              TABLES
                lines           = li_tdline.


            CLEAR : lwa_hdr_long_text, lwa_detail.
            CLEAR li_tdline.
          ENDLOOP.

        ENDIF.

*        lwa_log_h-belnr    = lv_objkey(10).

      ENDIF.
    ENDIF.

    result = NEW zcl_ava_log_repository=>ty_source(  ).
    ASSIGN result->* TO <result>.

    ASSIGN <source>-t_log_header[ 1 ] TO FIELD-SYMBOL(<log_header>).
    IF <log_header> IS ASSIGNED.

      "Populate header log
      <result>-t_log_header = VALUE #( (
      zsession  = <log_header>-zsession
      proc_user = sy-uname
      iffprocdt = sy-datum
      iffproctm = sy-uzeit
      zcount    = <log_header>-zcount
      zeline    = <log_header>-zeline
      zpath     = <log_header>-zpath
      bukrs     = <source>-document_header-comp_code
      belnr     = lv_objkey(10)
      bldat     = <source>-document_header-doc_date
      budat     = <source>-document_header-pstng_date
      xblnr     = <source>-document_header-ref_doc_no
      bktxt     = <source>-document_header-header_txt
      blart     = <source>-document_header-doc_type
      waers     = <log_header>-waers
      type      = lv_type
      message   = |{ <log_header>-message } { lv_message }| ) ).


    ENDIF.

    LOOP AT <source>-t_log_detail ASSIGNING FIELD-SYMBOL(<log_detail>).
      APPEND INITIAL LINE TO <result>-t_log_detail ASSIGNING FIELD-SYMBOL(<log_d_result>).
      <log_d_result> = CORRESPONDING #( <log_detail> ).
      <log_d_result>-type = lv_type.
      <log_d_result>-message = |{ <log_detail>-message } { lv_message }|.

    ENDLOOP.


  ENDMETHOD.

ENDCLASS.

          lwa_accountgl-material    = lwa_upload-matnr.

          IF lwa_documentheader-comp_code CP 'IN*'.
            lwa_accountgl-itemno_tax = '000001'.

          ENDIF.

            lwa_tax_sum-mwskz   = lwa_upload-mwskz.
            lwa_tax_sum-waers   = lv_waers.

              lwa_tax_sum-fwbas   = lwa_upload-fwbas.
            ENDIF.

          ENDIF.

          "No tax code, set full amount
        ELSE.
          lwa_currency-amt_doccur   = lwa_upload-wrbtr.

  METHOD adjust_tax.
    CONSTANTS lc_mwskz_a0 TYPE bseg-mwskz VALUE 'A0'.
    DATA li_tax_refkey      TYPE ztinvoice_gen_detail.
    DATA lv_tax_msg_type TYPE char1.
    DATA lv_mwskz TYPE bseg-mwskz.
    DATA: lv_xblnr_n       TYPE REF TO data,
          lv_xblnr_add_seq TYPE i.

    DATA(lv_tax_testrun) = COND abap_bool( WHEN im_execute = abap_true
                                           THEN abap_false
                                           ELSE abap_true ).

    DATA(lv_tax_commit) = COND abap_bool( WHEN im_execute = abap_true
                                           THEN abap_true
                                           ELSE abap_false ).

    REFRESH : li_tax_refkey.
    CLEAR   : lv_tax_msg_type, lv_mwskz.

    "Get Last Tax Code as last program Behavior
    LOOP AT im_t_upload INTO DATA(lwa_upload_tax) WHERE zcount EQ im_upload-zcount.
      IF lwa_upload_tax-mwskz IS NOT INITIAL.
        lv_mwskz = lwa_upload_tax-mwskz..
      ENDIF.
    ENDLOOP.

    IF lv_mwskz IS NOT INITIAL AND
       lv_mwskz <> lc_mwskz_a0.

      li_tax_refkey = VALUE #( ( bukrs = im_upload-bukrs
                                  gjahr = im_upload-budat(4)
                                  mwskz = lv_mwskz
                                ) ).

      CALL FUNCTION 'ZF_REF_NO_FROM_SUBSTITUTION'
        EXPORTING
          v_testrun         = lv_tax_testrun
          im_commit         = lv_tax_commit
        IMPORTING
          ex_message_status = lv_tax_msg_type
        CHANGING
          ch_refkey         = li_tax_refkey.

      IF lv_tax_msg_type EQ fiehc_con_msgty_e.

        ex_message = 'Tax Invoice Number Does Not Exist'.
        ex_mtype = c_message_e.

      ELSE.

        ASSIGN li_tax_refkey[ 1 ] TO FIELD-SYMBOL(<tax_refkey>).
        IF <tax_refkey> IS ASSIGNED.
          IF im_execute EQ abap_false. " Handle the Sequence for Test Run
            DATA(lv_refkey_len) = strlen( <tax_refkey>-xblnr ).
            CREATE DATA lv_xblnr_n TYPE n LENGTH lv_refkey_len.
*            FIELD-SYMBOLS <lv_xblnr_n> TYPE any.
            ASSIGN lv_xblnr_n->* TO FIELD-SYMBOL(<lv_xblnr_n>).
            <lv_xblnr_n> = <tax_refkey>-xblnr + lv_xblnr_add_seq.
            ex_xblnr = <lv_xblnr_n>.
            lv_xblnr_add_seq += 1.
          ELSE.
            ex_xblnr = <tax_refkey>-xblnr.
          ENDIF.

        ENDIF.
      ENDIF.

    ELSE.
      ex_message = 'Tax Code is Missing from Document'.
      ex_mtype   = c_message_e.
    ENDIF.

  ENDMETHOD.

  METHOD populate_tax_line.
    DATA lv_line_no TYPE i.
    DATA lwa_tax TYPE bapiactx09.
    DATA: lwa_currency TYPE bapiaccr09,
          lv_fwbas     TYPE bseg-fwbas,
          lv_fwste     TYPE bset-fwste,
          li_rtax1u15  TYPE STANDARD TABLE OF rtax1u15.

    "Populate tax line item grouped by tax code
    LOOP AT im_t_tax_sum INTO DATA(lwa_tax_sum).

      lv_line_no += 1.

      "Populate line item tax
      CLEAR lwa_tax.
      lwa_tax-itemno_acc = lv_line_no.
      lwa_tax-tax_code   = lwa_tax_sum-mwskz.

      READ TABLE im_t_a003 INTO DATA(lwa_a003) WITH KEY mwskz = lwa_tax_sum-mwskz BINARY SEARCH.
      IF sy-subrc = 0.
        lwa_tax-cond_key = lwa_a003-kschl.
      ENDIF.

*        "Avoid error 'Taxes by item is activated; consequently, transfer by item is mandatory'
      IF im_documentheader-comp_code CP 'IN*'.
        lwa_tax-itemno_tax = '000001'.
      ENDIF.

      APPEND lwa_tax TO result-t_tax.

      "Populate currency tax
      CLEAR lwa_currency.

      lwa_currency-itemno_acc   = lv_line_no.
      lwa_currency-curr_type    = c_curr_type_00.
      lwa_currency-currency     = im_waers.

      "IF tax calculated is 1%, using value from amount base for the calculation
      IF lwa_tax_sum-fwbas IS NOT INITIAL.
        lv_fwbas = lwa_tax_sum-fwbas.
      ELSE.
        lv_fwbas = lwa_tax_sum-wrbtr.
      ENDIF.

      CALL FUNCTION 'CALCULATE_TAX_FROM_NET_AMOUNT'
        EXPORTING
          i_bukrs           = im_documentheader-comp_code
          i_mwskz           = lwa_tax_sum-mwskz
          i_waers           = im_waers
          i_wrbtr           = lv_fwbas
        IMPORTING
          e_fwste           = lv_fwste
        TABLES

          bukrs_not_found   = 1
          country_not_found = 2
          mwskz_not_defined = 3
          mwskz_not_valid   = 4
          ktosl_not_found   = 5
          kalsm_not_found   = 6
          parameter_error   = 7
          knumh_not_found   = 8
          kschl_not_found   = 9
          unknown_error     = 10
          account_not_found = 11
          txjcd_not_valid   = 12
          OTHERS            = 13.
      IF sy-subrc <> 0.
* Implement suitable error handling here
      ENDIF.

      READ TABLE li_rtax1u15 INTO DATA(lwa_rtax1u15) INDEX 1.
      IF sy-subrc = 0.
        IF im_waers IN im_r_round_curr.
          lwa_currency-amt_doccur   = lwa_rtax1u15-wmwst * 100.
          lwa_currency-amt_base     = lv_fwbas * 100.
          lwa_currency-tax_amt      = lv_fwste * 100.
        ELSE.
          lwa_currency-amt_doccur   = lwa_rtax1u15-wmwst.
          lwa_currency-amt_base     = lv_fwbas.
          lwa_currency-tax_amt      = lv_fwste.

  METHOD read_csv_file.

    lc_max_csv_row TYPE i VALUE '1000000'.

*    DELETE li_string INDEX 1.

  METHOD run_background.
    CONSTANTS c_jobname TYPE tbtcjob-jobname VALUE 'ZAVALON'.
    CONSTANTS c_program_name TYPE char25 VALUE 'ZFI_AVALON_JOURNAL_POST'.
    DATA lv_jobname TYPE btcjob.
    DATA lv_session TYPE zfava_post_stg-zsession.

    DATA:
      lv_jobcount TYPE tbtcjob-jobcount.

    IF im_session IS INITIAL.
      TRY.
          DATA(lt_source) = get_source_data(
                              im_session  = im_session
                              im_filepath = im_filepath
                              im_bjob     = im_bjob
                            ).
          ASSIGN lt_source[ 1 ]-zsession TO FIELD-SYMBOL(<sess>).
          IF <sess> IS ASSIGNED.
            lv_session = <sess>.
          ENDIF.
        CATCH zcx_data_service.
          "handle exception
      ENDTRY.
    ELSE.
      lv_session = im_session.
    ENDIF.

    DATA(lv_session_text) = lv_session.

    SHIFT lv_session_text LEFT DELETING LEADING '0'.

    lv_jobname = |{ c_jobname }_{ lv_session_text }|.


    CALL FUNCTION 'JOB_OPEN'
      EXPORTING
        jobname          = lv_jobname
      IMPORTING
        jobcount         = lv_jobcount
      EXCEPTIONS
        cant_create_job  = 1
        invalid_job_data = 2
        jobname_missing  = 3
        OTHERS           = 4.

    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
        WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ELSE.
      DATA(lv_test) = COND abap_bool( WHEN im_execute = abap_true
                                      THEN abap_false
                                      ELSE abap_true ).

      SUBMIT (c_program_name) AND RETURN

CLASS lcl_template_downloader IMPLEMENTATION.

  METHOD download.


    DATA lv_dir         TYPE rlgrap-filename.
    DATA ls_wwwdata_tab TYPE wwwdatatab.
*    DATA lv_message     TYPE string.

*Make sure template exists in SMw0 as binary object
    SELECT * FROM wwwdata INNER JOIN tadir
                  ON wwwdata~objid = tadir~obj_name
                  INTO  CORRESPONDING FIELDS OF ls_wwwdata_tab UP TO 1 ROWS
                  WHERE wwwdata~srtf2   = 0
                  AND   wwwdata~relid   = 'MI'
                  AND   tadir~pgmid     = 'R3TR'
                  AND   tadir~object    = 'W3MI'
                  AND   tadir~obj_name  = 'ZXLWB_ZFE_MASS_JOURNAL_UPLOAD'.
    ENDSELECT.

    lv_dir = get_directory(  ).
    IF lv_dir IS NOT INITIAL.

      lv_dir = |{ lv_dir }/Mass_Journal_Template.xlsx|.

      CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
        EXPORTING
          key         = ls_wwwdata_tab
          destination = lv_dir.
      IF sy-subrc EQ 0.
        result-message = |Successfully download template { lv_dir }|.
        result-type = 'S'.
      ENDIF.

    ELSE.
      result-message = 'An error has occured picking a folder'.
      result-type = 'E'.

    ENDIF.


  ENDMETHOD.


  METHOD get_directory.

    DATA: objfile TYPE REF TO cl_gui_frontend_services.
    DATA: pickedfolder TYPE string.
    DATA: initialfolder TYPE string.

    IF sy-batch IS INITIAL.
      CREATE OBJECT objfile.

      objfile->get_temp_directory( CHANGING   temp_dir              = initialfolder
                                   EXCEPTIONS cntl_error            = 1
                                              error_no_gui          = 2
                                              not_supported_by_gui  = 3 ).

      objfile->directory_browse( EXPORTING  initial_folder        = initialfolder
                                 CHANGING   selected_folder       = pickedfolder
                                 EXCEPTIONS cntl_error            = 1
                                            error_no_gui          = 2
                                            not_supported_by_gui  = 3 ).

      IF sy-subrc = 0.
        result = pickedfolder.
      ENDIF.
    ENDIF.

  ENDMETHOD.
