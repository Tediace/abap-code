  METHOD lif_events_helper~modify_cell_data_changed.
    DATA credit TYPE hslxx.
    DATA debit TYPE hslxx.
    DATA unaudited_bal TYPE hslxx.
    DATA unaudited_bal_after_adj TYPE hslxx.
    DATA is_permanent TYPE abap_bool.
    DATA formula TYPE zde_cit_formula.
    DATA permanent TYPE hslxx12.
    DATA timing TYPE hslxx12.
    DATA depre_val TYPE hslxx12.

    " modify value
    SELECT DISTINCT row_id,
                    tabix
    FROM @mr_data_changed->mt_mod_cells AS s
    INTO TABLE @DATA(row_ids)
    ##ITAB_DB_SELECT
    ##ITAB_KEY_IN_SELECT.

    LOOP AT row_ids REFERENCE INTO DATA(row_id).
      CLEAR: credit, debit, unaudited_bal, unaudited_bal_after_adj, depre_val.
      ASSIGN mr_data_changed->mt_mod_cells[ row_id = row_id->row_id
                                            tabix = row_id->tabix
                                            fieldname = 'ADJUSTMENT_CREDIT' ]
                                            TO FIELD-SYMBOL(<credit>).
      IF <credit> IS ASSIGNED.
        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = <credit>-row_id
            i_tabix     = <credit>-tabix
            i_fieldname = <credit>-fieldname
          IMPORTING
            e_value     = credit
        ).

      ELSE.
        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'ADJUSTMENT_CREDIT'
          IMPORTING
            e_value     = credit
        ).


      ENDIF.

      ASSIGN mr_data_changed->mt_mod_cells[ row_id = row_id->row_id
                                            tabix = row_id->tabix
                                            fieldname = 'ADJUSTMENT_DEBIT' ]
                                            TO FIELD-SYMBOL(<debit>).
      IF <debit> IS ASSIGNED.
        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = <debit>-row_id
            i_tabix     = <debit>-tabix
            i_fieldname = <debit>-fieldname
          IMPORTING
            e_value     = debit
        ).

      ELSE.

        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'ADJUSTMENT_DEBIT'
          IMPORTING
            e_value     = debit
        ).
      ENDIF.

      mr_data_changed->get_cell_value(
        EXPORTING
          i_row_id    = row_id->row_id
          i_tabix     = row_id->tabix
          i_fieldname = 'UNAUDITED_BALANCE'
        IMPORTING
          e_value     = unaudited_bal
      ).
      " calculate
      unaudited_bal_after_adj = unaudited_bal + debit + credit.
      " modify
      mr_data_changed->modify_cell(
        EXPORTING
          i_row_id    = row_id->row_id
          i_tabix     = row_id->tabix
          i_fieldname = 'AUDITED_BALANCE'
          i_value     = unaudited_bal_after_adj
      ).

      mr_data_changed->get_cell_value(
      EXPORTING
        i_row_id    = row_id->row_id
        i_tabix     = row_id->tabix
        i_fieldname = 'CORRECTION_FORMULA'
      IMPORTING
        e_value     = formula
    ).

      " modify fiscal correction with formula
      IF formula IS NOT INITIAL.
        mr_data_changed->get_cell_value(
      EXPORTING
        i_row_id    = row_id->row_id
        i_tabix     = row_id->tabix
        i_fieldname = 'INITIAL_DEPRE'
      IMPORTING
        e_value     = depre_val
      ).

        DATA(fiscal_correction) = get_fiscal_correction(
                                    unaudited_bal_after_adj = unaudited_bal_after_adj
                                    depre_val               = depre_val
                                    formula                 = formula
                                  ).


        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'IS_CORRECTION_PERMANENT'
          IMPORTING
            e_value     = is_permanent
        ).

        IF is_permanent = abap_true.
          mr_data_changed->modify_cell(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
            i_value     = fiscal_correction
        ).

        ELSE.
          mr_data_changed->modify_cell(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'FISCAL_CORRECTION_TIMING'
            i_value     = fiscal_correction
        ).

        ENDIF.

        " fiscal balance
        mr_data_changed->modify_cell(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'FISCAL_BALANCE'
            i_value     = unaudited_bal_after_adj + fiscal_correction
        ).
      ELSE.
        CLEAR: permanent, timing.
        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
          IMPORTING
            e_value     = permanent
        ).
        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'FISCAL_CORRECTION_TIMING'
          IMPORTING
            e_value     = timing
        ).

        " fiscal balance
        mr_data_changed->modify_cell(
          EXPORTING
            i_row_id    = row_id->row_id
            i_tabix     = row_id->tabix
            i_fieldname = 'FISCAL_BALANCE'
            i_value     = unaudited_bal_after_adj + permanent + timing
        ).

      ENDIF.
      UNASSIGN: <debit>, <credit>.
    ENDLOOP.

  ENDMETHOD.

  METHOD lif_events_helper~process_manual_fisc_correct.
    DATA permanent_cell_val TYPE hslxx12.
    DATA timing_cell_val TYPE hslxx12.

    SELECT *
    FROM @mr_data_changed->mt_mod_cells AS s
    WHERE fieldname = 'FISCAL_CORRECTION_PERMANENT'
    INTO TABLE @DATA(mod_permanents)
    ##ITAB_KEY_IN_SELECT.

    SELECT *
    FROM @mr_data_changed->mt_mod_cells AS s
    WHERE fieldname = 'FISCAL_CORRECTION_TIMING'
    INTO TABLE @DATA(mod_timings)
    ##ITAB_KEY_IN_SELECT.


    LOOP AT mod_permanents REFERENCE INTO DATA(permanent).
      CLEAR permanent_cell_val.

      mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = permanent->row_id
            i_tabix     = permanent->tabix
            i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
          IMPORTING
            e_value     = permanent_cell_val
        ).
      ASSIGN mr_data_changed->mt_mod_cells[ row_id = permanent->row_id
                                            tabix = permanent->tabix
                                            fieldname = 'FISCAL_CORRECTION_TIMING' ]
                                            TO FIELD-SYMBOL(<timing>).

      IF <timing> IS ASSIGNED.
        CLEAR timing_cell_val.
        " check when performing clearing on at least one of the cell value
        " permanent value --> initial
        " or
        " timing value --> initial
        " or
        " both --> initial

        mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = permanent->row_id
            i_tabix     = permanent->tabix
            i_fieldname = 'FISCAL_CORRECTION_TIMING'
          IMPORTING
            e_value     = timing_cell_val
        ).
        IF permanent_cell_val IS NOT INITIAL
          AND timing_cell_val IS NOT INITIAL. " when both cells are occupied
          " error

          mr_data_changed->add_protocol_entry(
            EXPORTING
              i_msgid     = 'DB'
              i_msgty     = 'E'
              i_msgno     = '711'
              i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
              i_row_id    = permanent->row_id
              i_tabix     = permanent->tabix
          ).

          mr_data_changed->add_protocol_entry(
            EXPORTING
              i_msgid     = 'DB'
              i_msgty     = 'E'
              i_msgno     = '711'
              i_fieldname = 'FISCAL_CORRECTION_TIMING'
              i_row_id    = permanent->row_id
              i_tabix     = permanent->tabix
          ).

        ELSEIF permanent_cell_val IS INITIAL
           AND timing_cell_val IS NOT INITIAL.
          mr_data_changed->modify_style(
            EXPORTING
              i_row_id    = permanent->row_id
              i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
              i_style     = cl_gui_alv_grid=>mc_style_disabled
          ).
          modify_manual_fisc_balance(
          EXPORTING
            row_id          = permanent->row_id
            tabix           = permanent->tabix
            fisc_correction = permanent_cell_val
        ).

        ELSE.
          mr_data_changed->modify_style(
              EXPORTING
                i_row_id    = permanent->row_id
                i_fieldname = 'FISCAL_CORRECTION_TIMING'
                i_style     = cl_gui_alv_grid=>mc_style_disabled
            ).
          modify_manual_fisc_balance(
          EXPORTING
            row_id          = permanent->row_id
            tabix           = permanent->tabix
            fisc_correction = permanent_cell_val
        ).

        ENDIF.

      ELSE.
        IF permanent_cell_val IS NOT INITIAL.
          mr_data_changed->modify_style(
            EXPORTING
              i_row_id    = permanent->row_id
              i_fieldname = 'FISCAL_CORRECTION_TIMING'
              i_style     = cl_gui_alv_grid=>mc_style_disabled
          ).

        ELSE.
*          mr_data_changed->modify_style(
*            EXPORTING
*              i_row_id    = permanent->row_id
*              i_fieldname = 'FISCAL_CORRECTION_TIMING'
*              i_style     = cl_gui_alv_grid=>mc_style_enabled
*          ).

          mr_data_changed->get_cell_value(
          EXPORTING
            i_row_id    = permanent->row_id
            i_tabix     = permanent->tabix
            i_fieldname = 'FISCAL_CORRECTION_TIMING'
          IMPORTING
            e_value     = timing_cell_val
        ).
          IF <timing> IS ASSIGNED.
            mr_data_changed->modify_style(
              EXPORTING
                i_row_id    = permanent->row_id
                i_fieldname = 'FISCAL_CORRECTION_TIMING'
                i_style     = cl_gui_alv_grid=>mc_style_enabled
            ).
          ELSEIF timing_cell_val IS NOT INITIAL.
            mr_data_changed->modify_style(
            EXPORTING
              i_row_id    = permanent->row_id
              i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
              i_style     = cl_gui_alv_grid=>mc_style_disabled ).

          ENDIF.
*          IF timing_cell_val IS NOT INITIAL.
*            mr_data_changed->modify_style(
*            EXPORTING
*              i_row_id    = permanent->row_id
*              i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
*              i_style     = cl_gui_alv_grid=>mc_style_disabled
*          ).
*          ENDIF.

        ENDIF.

        " fiscal balance
        modify_manual_fisc_balance(
          EXPORTING
            row_id          = permanent->row_id
            tabix           = permanent->tabix
            fisc_correction = permanent_cell_val
        ).
      ENDIF.

    ENDLOOP.

    LOOP AT mod_timings REFERENCE INTO DATA(timing).
      ASSIGN mr_data_changed->mt_mod_cells[ row_id = timing->row_id
                                            tabix = timing->tabix
                                            fieldname = 'FISCAL_CORRECTION_PERMANENT' ]
                                            TO FIELD-SYMBOL(<permanent>).
      CLEAR timing_cell_val.
      " when perform clearing one of the cell

      mr_data_changed->get_cell_value(
        EXPORTING
          i_row_id    = timing->row_id
          i_tabix     = timing->tabix
          i_fieldname = 'FISCAL_CORRECTION_TIMING'
        IMPORTING
          e_value     = timing_cell_val
      ).
      IF <permanent> IS NOT ASSIGNED.
        IF timing_cell_val IS NOT INITIAL.
          mr_data_changed->modify_style(
            EXPORTING
              i_row_id    = timing->row_id
              i_fieldname = 'FISCAL_CORRECTION_PERMANENT'

              i_fieldname = 'FISCAL_CORRECTION_PERMANENT'
              i_style     = cl_gui_alv_grid=>mc_style_enabled

          mr_data_changed->get_cell_value(
        EXPORTING
          i_row_id    = timing->row_id
          i_tabix     = timing->tabix

            mr_data_changed->modify_style(
            EXPORTING
              i_row_id    = timing->row_id
              i_fieldname = 'FISCAL_CORRECTION_TIMING'
              i_style     = cl_gui_alv_grid=>mc_style_disabled
          ).
          ENDIF.


        ENDIF.

        " fiscal balance

        modify_manual_fisc_balance(
          EXPORTING
            row_id          = timing->row_id
            tabix           = timing->tabix
            fisc_correction = timing_cell_val
        ).
      ENDIF.

    ENDLOOP.

  ENDMETHOD.

FORM exit_0100 .
  DATA l_okcode LIKE sy-ucomm.

  l_okcode = g_okcode.
  CLEAR g_okcode.

  CASE l_okcode.
    WHEN con_exit OR con_back OR con_canc.
      gr_grid_d0100->free( ).
      gr_container_d0100->free( ).
      cl_gui_cfw=>flush( ).

      CLEAR gr_container_d0100.
      CLEAR gr_grid_d0100.
      CLEAR gr_events_d0100.

      SET SCREEN 0.
      LEAVE SCREEN.
  ENDCASE.

ENDFORM.

*& text
*&---------------------------------------------------------------------*
FORM fcode_0100 .

  DATA: l_okcode LIKE sy-ucomm.

  l_okcode = g_okcode.
  CLEAR g_okcode.

  CASE l_okcode.
    WHEN con_exit OR con_back OR con_canc.
      g_okcode = l_okcode.
      PERFORM exit_0100.

    WHEN 'TOGGLE'.                                          "#EC NOTEXT
      DATA: l_edit TYPE i.

      CASE gs_spec-edit_mode.
        WHEN con_edit_mode_change.
          gs_spec-edit_mode = con_edit_mode_display.
          l_edit            = con_off.
        WHEN con_edit_mode_display.
          gs_spec-edit_mode = con_edit_mode_change.
          l_edit            = con_on.
      ENDCASE.

      gr_grid_d0100->set_ready_for_input( l_edit ).
  ENDCASE.

ENDFORM.

FORM set_header_0100.
  gr_docking_container = NEW cl_gui_docking_container(
    repid                   = sy-repid
    dynnr                   = '0100'
    ratio                   = 15
    side                    = cl_gui_docking_container=>dock_at_top
  ).

  gr_document = NEW cl_dd_document( ).

  SELECT SINGLE a~name1
  FROM t001 AS t
  INNER JOIN adrc AS a ON t~bukrs = @p_bukrs
                      AND t~adrnr = a~addrnumber
   INTO @DATA(name).

  gr_document->add_text(
    EXPORTING
      text_table    = VALUE #( ( name )
                               ( 'Corporate Income Tax Calculation' )
                               ( |FY{ p_gjahr }| ) )
      sap_emphasis  = cl_dd_document=>strong
      sap_fontsize  = cl_dd_document=>medium
      fix_lines = 'X'
  ).

  gr_document->display_document(
    EXPORTING
      parent             = gr_docking_container
  ).

ENDFORM.

FORM print_top_of_list.

  SELECT SINGLE a~name1
  FROM t001 AS t
  INNER JOIN adrc AS a ON t~bukrs = @p_bukrs
                      AND t~adrnr = a~addrnumber
   INTO @DATA(name).

  WRITE name && ' Corporate Income Tax Calculation' && ' FY' && p_gjahr.


ENDFORM.
